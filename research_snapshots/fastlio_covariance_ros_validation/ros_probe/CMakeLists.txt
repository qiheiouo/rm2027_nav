cmake_minimum_required(VERSION 3.8)
project(rm_fastlio_covariance_ros_validation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(fast_lio_multi REQUIRED)
foreach(dep rclcpp std_msgs sensor_msgs geometry_msgs nav_msgs tf2 tf2_ros tf2_geometry_msgs tf2_eigen pcl_conversions livox_ros_driver2)
  find_package(${dep} REQUIRED)
endforeach()
get_filename_component(FASTLIO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/fast_lio_multi" ABSOLUTE)
get_filename_component(FASTLIO_PREFIX "${fast_lio_multi_DIR}/../../.." ABSOLUTE)
find_library(FASTLIO_PREPROCESS fastlio_preprocess PATHS "${FASTLIO_PREFIX}/lib/fast_lio_multi" NO_DEFAULT_PATH REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  "${FASTLIO_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
foreach(mode bundle adaptive async)
  add_executable(covariance_${mode} publication_boundary.cpp)
  target_compile_definitions(covariance_${mode} PRIVATE
    ROOT_DIR="${FASTLIO_SOURCE_DIR}/"
    FASTLIO_SOURCE_FILE="${FASTLIO_SOURCE_DIR}/src/laserMapping_${mode}.cpp")
  if(mode STREQUAL "bundle")
    target_compile_definitions(covariance_${mode} PRIVATE COVARIANCE_BUNDLE)
  elseif(mode STREQUAL "adaptive")
    target_compile_definitions(covariance_${mode} PRIVATE FASTLIO_ADAPTIVE)
  endif()
  target_include_directories(covariance_${mode} PRIVATE "${FASTLIO_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_BINARY_DIR}" ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS})
  target_compile_options(covariance_${mode} PRIVATE -O0)
  target_link_libraries(covariance_${mode} ${FASTLIO_PREPROCESS} ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
  ament_target_dependencies(covariance_${mode} rclcpp std_msgs sensor_msgs geometry_msgs nav_msgs
    tf2 tf2_ros tf2_geometry_msgs tf2_eigen pcl_conversions livox_ros_driver2)
  install(TARGETS covariance_${mode} DESTINATION lib/${PROJECT_NAME})
endforeach()
find_package(rm_competition_interfaces REQUIRED)
find_package(rm_localization_adapters REQUIRED)
get_filename_component(ADAPTER_PREFIX "${rm_localization_adapters_DIR}/../../.." ABSOLUTE)
find_library(CANONICAL_ODOMETRY canonical_odometry PATHS "${ADAPTER_PREFIX}/lib/rm_localization_adapters" NO_DEFAULT_PATH REQUIRED)
get_filename_component(BASELINE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/covariance_baseline_source/lio_adapter.cpp" ABSOLUTE)
if(EXISTS "${BASELINE_SOURCE}")
  add_executable(baseline_lio_adapter "${BASELINE_SOURCE}")
  target_include_directories(baseline_lio_adapter PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../src/rm_localization_adapters/include")
  target_link_libraries(baseline_lio_adapter ${CANONICAL_ODOMETRY})
  ament_target_dependencies(baseline_lio_adapter rclcpp rm_competition_interfaces nav_msgs geometry_msgs
    builtin_interfaces tf2 tf2_ros tf2_geometry_msgs)
  install(TARGETS baseline_lio_adapter DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
