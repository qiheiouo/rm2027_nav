cmake_minimum_required(VERSION 3.16)
project(rm_full_base_shadow_validation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(rm_shadow_observations SHARED src/observations.cpp)
target_compile_features(rm_shadow_observations PUBLIC cxx_std_17)
target_compile_options(rm_shadow_observations PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rm_shadow_observations gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS rm_shadow_observations LIBRARY DESTINATION lib)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/ingress.py scripts/candidate.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/adapter.py scripts/common.py scripts/input_contract.py DESTINATION lib/${PROJECT_NAME})
ament_package()
