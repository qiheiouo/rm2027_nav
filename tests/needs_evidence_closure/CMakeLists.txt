cmake_minimum_required(VERSION 3.16)
project(needs_evidence_closure LANGUAGES C CXX)
enable_testing()
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenMP REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT EXISTS "${FASTLIO_SOURCE}/src/laserMapping_bundle.cpp")
  message(FATAL_ERROR "Set FASTLIO_SOURCE to the unchanged pinned dependency")
endif()
# Reuse the established visibility-only preparation, compiling all actual methods.
execute_process(COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/../final_dual_closure/prepare_visible_bundle.py
  ${FASTLIO_SOURCE}/src/laserMapping_bundle.cpp ${CMAKE_CURRENT_BINARY_DIR}/bundle_prepared.cpp
  COMMAND_ERROR_IS_FATAL ANY)
configure_file(${CMAKE_CURRENT_BINARY_DIR}/bundle_prepared.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/bundle_visible.cpp COPYONLY)
foreach(fixture bundle deskew)
  add_executable(${fixture}_evidence ${fixture}_evidence.cpp)
  target_compile_features(${fixture}_evidence PRIVATE cxx_std_17)
  target_compile_options(${fixture}_evidence PRIVATE -O0 -g0 -UNDEBUG -D_GLIBCXX_ASSERTIONS)
  target_compile_definitions(${fixture}_evidence PRIVATE
    ROOT_DIR="${FASTLIO_SOURCE}/" BUNDLE_VISIBLE="${CMAKE_CURRENT_BINARY_DIR}/bundle_visible.cpp")
  target_include_directories(${fixture}_evidence PRIVATE ${FASTLIO_SOURCE} ${FASTLIO_SOURCE}/src ${FASTLIO_SOURCE}/include ${PCL_INCLUDE_DIRS})
  target_link_libraries(${fixture}_evidence ${FASTLIO_LIBRARY} ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
  ament_target_dependencies(${fixture}_evidence rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
endforeach()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(mode imu lidar dual fov)
  add_test(NAME aud031_040_${mode}_characterization COMMAND bundle_evidence ${mode}
    --ros-args --params-file ${REPO_ROOT}/src/rm_lio_bringup/config/fast_lio_multi_single_mid360.yaml)
  set_tests_properties(aud031_040_${mode}_characterization PROPERTIES TIMEOUT 20)
endforeach()
add_test(NAME aud052_known_deskew_error_characterization COMMAND deskew_evidence)
set_tests_properties(aud052_known_deskew_error_characterization PROPERTIES TIMEOUT 20)
add_executable(livox_phase_evidence livox_phase_evidence.cpp ${LIVOX_SOURCE}/src/src/comm/pub_handler.cpp)
target_compile_features(livox_phase_evidence PRIVATE cxx_std_17)
target_compile_options(livox_phase_evidence PRIVATE -UNDEBUG)
target_include_directories(livox_phase_evidence PRIVATE ${LIVOX_SOURCE}/src/src /usr/local/include)
target_link_libraries(livox_phase_evidence pthread)
add_test(NAME aud054_phase_characterization COMMAND livox_phase_evidence)
set_tests_properties(aud054_phase_characterization PROPERTIES TIMEOUT 20)
foreach(dep rclcpp_action nav2_msgs nav2_behavior_tree behaviortree_cpp_v3)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(bt_late_accept_evidence bt_late_accept_evidence.cpp)
target_compile_features(bt_late_accept_evidence PRIVATE cxx_std_17)
target_compile_options(bt_late_accept_evidence PRIVATE -UNDEBUG)
ament_target_dependencies(bt_late_accept_evidence rclcpp rclcpp_action nav2_msgs nav2_behavior_tree behaviortree_cpp_v3)
add_test(NAME aud056_late_accept_characterization COMMAND bt_late_accept_evidence)
set_tests_properties(aud056_late_accept_characterization PROPERTIES TIMEOUT 15)
add_test(NAME aud016_planner_cancel_characterization COMMAND
  ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/planner_cancel_evidence.py)
set_tests_properties(aud016_planner_cancel_characterization PROPERTIES
  TIMEOUT 30 RUN_SERIAL TRUE ENVIRONMENT "EVIDENCE_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}/planner")
