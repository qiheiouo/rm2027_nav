cmake_minimum_required(VERSION 3.16)
project(confirmed_software_fixes LANGUAGES CXX)
enable_testing()
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(cancel_wait_fixture SHARED cancel_wait_fixture.cpp)
ament_target_dependencies(cancel_wait_fixture nav2_core nav2_costmap_2d pluginlib std_msgs std_srvs)
pluginlib_export_plugin_description_file(nav2_core fixture_controller.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d fixture_layer.xml)
install(TARGETS cancel_wait_fixture LIBRARY DESTINATION lib)
foreach(mode pose poses controller)
  add_test(NAME aud016_${mode} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cancel_wait_regression.py ${mode})
  set_tests_properties(aud016_${mode} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE ENVIRONMENT "EVIDENCE_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}/${mode}")
endforeach()

if(DEFINED FASTLIO_SOURCE)
  foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
    find_package(${dep} REQUIRED)
  endforeach()
  find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
  find_package(OpenMP REQUIRED)
  add_executable(deskew_regression deskew_regression.cpp)
  target_compile_features(deskew_regression PRIVATE cxx_std_17)
  target_compile_options(deskew_regression PRIVATE -O0 -g0 -UNDEBUG -D_GLIBCXX_ASSERTIONS)
  target_compile_definitions(deskew_regression PRIVATE ROOT_DIR="${FASTLIO_SOURCE}/")
  target_include_directories(deskew_regression PRIVATE ${FASTLIO_SOURCE} ${FASTLIO_SOURCE}/include ${PCL_INCLUDE_DIRS})
  target_link_libraries(deskew_regression ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
  ament_target_dependencies(deskew_regression rclcpp sensor_msgs nav_msgs geometry_msgs tf2 tf2_ros tf2_eigen tf2_geometry_msgs pcl_conversions livox_ros_driver2)
  add_test(NAME aud052_deskew COMMAND deskew_regression)
  set_tests_properties(aud052_deskew PROPERTIES TIMEOUT 20)
endif()

foreach(dep rclcpp_action nav2_msgs nav2_behavior_tree behaviortree_cpp_v3)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(bt_late_goal_regression bt_late_goal_regression.cpp)
target_compile_features(bt_late_goal_regression PRIVATE cxx_std_17)
if(DEFINED NAV2_BT_SOURCE)
  target_include_directories(bt_late_goal_regression BEFORE PRIVATE ${NAV2_BT_SOURCE}/include)
endif()
target_compile_options(bt_late_goal_regression PRIVATE -UNDEBUG)
ament_target_dependencies(bt_late_goal_regression rclcpp rclcpp_action nav2_msgs nav2_behavior_tree behaviortree_cpp_v3)
foreach(mode timeout halt reject refuse destroy known update completed shutdown)
  add_test(NAME aud056_${mode} COMMAND bt_late_goal_regression ${mode})
  set_tests_properties(aud056_${mode} PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
endforeach()

if(DEFINED NAV2_SPIN_PLUGIN)
  add_test(NAME aud056_plugin COMMAND bt_late_goal_regression plugin)
  set_tests_properties(aud056_plugin PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE ENVIRONMENT "NAV2_SPIN_PLUGIN=${NAV2_SPIN_PLUGIN}")
endif()

ament_package()
